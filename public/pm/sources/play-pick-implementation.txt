112: def calculate_preference_vector(user):
113:     """
114:     Calculate user preference vector from signals
115:
116:     Algorithm:
117:     1. Get all user signals
118:     2. For each signal, get performance embedding vector
119:     3. Weighted sum: sum(signal_weight * embedding_vector)
120:     4. L2 normalize result
121:
122:     Returns:
123:         1536-dim list of floats (normalized vector)
124:     """
125:     # Get user signals with embeddings
126:     signals = UserPerformanceSignal.objects.filter(
127:         user=user,
128:         performance__embedding__isnull=False
129:     ).select_related('performance__embedding')
130:
131:     if signals.count() == 0:
132:         # No signals: return zero vector
133:         return [0.0] * 1536
134:
135:     # Initialize accumulator
136:     weighted_sum = np.zeros(1536, dtype=np.float32)
137:
138:     for signal_obj in signals:
139:         embedding_vector = np.array(
140:             signal_obj.performance.embedding.vector,
141:             dtype=np.float32
142:         )
143:
144:         weight = signal_obj.signal
145:         weighted_sum += weight * embedding_vector
146:
147:     # L2 normalization
148:     norm = np.linalg.norm(weighted_sum)
149:     if norm > 0:
150:         normalized = weighted_sum / norm
151:     else:
152:         normalized = weighted_sum  # All zeros
153:
154:     return normalized.tolist()
155:
